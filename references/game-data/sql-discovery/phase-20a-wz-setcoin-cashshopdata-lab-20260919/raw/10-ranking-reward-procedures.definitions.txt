WZ_SetRankingDay	

CREATE Procedure [dbo].[WZ_SetRankingDay] 
@Value1 int,
@Value2 int,
@Value3 int,
@Value4 int,
@Value5 int,
@Value6 int,
@Value7 int,
@Value8 int,
@Value9 int
AS
BEGIN

SET NOCOUNT ON
SET XACT_ABORT ON

DECLARE @Exist varchar(10)
DECLARE @AccountID1 varchar(10)
DECLARE @AccountID2 varchar(10)
DECLARE @AccountID3 varchar(10)

SELECT TOP 1 @AccountID1=AccountID FROM Character ORDER BY ResetDay DESC
SELECT TOP 1 @AccountID2=AccountID FROM Character WHERE AccountID <> @AccountID1 ORDER BY ResetDay DESC
SELECT TOP 1 @AccountID3=AccountID FROM Character WHERE AccountID <> @AccountID1 AND AccountID <> @AccountID2 ORDER BY ResetDay DESC

-- Tipo 1: Update Cash | WCoinC
IF @Value1 > 0 OR @Value2 > 0 OR @Value3 > 0 
BEGIN
	--Altere de acordo com sua versão
	--Para Season 6/8
	UPDATE CashShopData SET WCoinC = WCoinC + @Value1 WHERE AccountID = @AccountID1 --1 do rank
	UPDATE CashShopData SET WCoinC = WCoinC + @Value2 WHERE AccountID = @AccountID2 --2 do rank
	UPDATE CashShopData SET WCoinC = WCoinC + @Value3 WHERE AccountID = @AccountID3 --3 do rank
	--Para Season 4
	--UPDATE MEMB_INFO SET Cash = Cash + @Value1 WHERE memb___id = @Account 
END

-- Tipo 1: Update Gold | WCoinP
IF @Value4 > 0 OR @Value5 > 0 OR @Value6 > 0 
BEGIN
	--Altere de acordo com sua versão
	--Para Season 6/8
	UPDATE CashShopData SET WCoinP = WCoinP + @Value4 WHERE AccountID = @AccountID1 --1 do rank
	UPDATE CashShopData SET WCoinP = WCoinP + @Value5 WHERE AccountID = @AccountID2 --2 do rank
	UPDATE CashShopData SET WCoinP = WCoinP + @Value6 WHERE AccountID = @AccountID3 --3 do rank
	--Para Season 4
	--UPDATE MEMB_INFO SET Cash = Cash + @Value1 WHERE memb___id = @Account

-- Tipo 1: Update PcPoint | Goblin Point

BEGIN

	--Para Season 6/8
	UPDATE CashShopData SET GoblinPoint = GoblinPoint + @Value7 WHERE AccountID = @AccountID1 --1 do rank

	UPDATE CashShopData SET GoblinPoint = GoblinPoint + @Value9 WHERE AccountID = @AccountID3 --3 do rank
